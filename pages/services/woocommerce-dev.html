// High Performance Order Storage (HPOS) compliant deposit splitting engine
final class DepositSplitEngine {
public function splitOrder(WC_Order $order, float $depositPercent): ScheduleResult {
$initialCharge = $order->get_total() * ($depositPercent / 100);
$balanceDue = $order->get_total() - $initialCharge;
$this->scheduler->scheduleBalanceCollection($order->get_id(), $balanceDue);
return ScheduleResult::success($initialCharge, $balanceDue);
}
}
